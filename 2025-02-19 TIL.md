```jsx
// $() : jQuery 객체 (문자열로 된 selector or Document Object)
// 전체 선택자 : * => 모든 element
// 타입 선택자 : 원하는 tag 명을 이용하여 선택
// 아이디 선택자 : # => 원하는 id를 가진 element 선택
// 클래스 선택자 : . => 원하는 class를 가진 element 선택
// 자식 선택자 : >
// 후손 선택자 : 공백
// 동위 선택자 : + (바로 다음만), ~ (바로 다음부터)
// 속성 선택자 : []
function myFunc() {
    // 타입 선택자
    $("h1, div").css("color", "red");
    // 아이디 선택자
    $("#busan").remove();
    // 클래스 선택자
    $(".myClass").css("background-color", "blue");
    // 자식 선택자
    $("ol > li").css("color", "yellow");
    // 후손 선택자
    $("div li").css("background-color", "green");
    // 동위 선택자 +
    $("#kang + li").css("font-weight", "bold");
    // 동위 선택자 ~
    alert($("#kang ~ li").text());
}

function selectValueChange() {
    let before = $("#regionOutput").val();
    let region = $("select > option:selected").text();
    $("#regionOutput").val(region);
    console.log(`changed! ${before} -> ${region}`);
}

function printSelected() {
    let checked = $("input[type=checkbox]:checked + span");
    $("#region").val(checked.text());
    checked.each(function (idx, item) {
        // idx : 0,1,2, ... for 문의 첨자 역할
        // item : 문서 객체 (jQuery 객체 X)
        console.log($(item).text() + "가 선택되었어요!");
    })
    $("input[type=checkbox]:checked").prop("checked", false);
}

function addPerson() {
    let newName = $("#name").val();
    let li = $("<li></li>").text(newName);
    // 1. append() : 제일 마지막 자식으로 붙여넣기
    // $("#names").append(li);
    // 2. prepend() : 제일 처음 자식으로 붙여넣기
    // $("#names").prepend(li);
    // 3. after() : 바로 다음 형제로 붙여넣기
    // $("#names > li:first").after(li);
    // 4. before() : 바로 앞 형제로 붙여넣기
    // $("#names > li:last").before(li);

    // 이미지 생성
    let newImage = $("<img />").attr("src", "img/car.jpg").attr("width", "200");
    $("#name").after(newImage);
}

function makeEvent() {
    $("h1").on("click", function () {
        // this의 원래 의미는 현재 사용되는 객체에 대한 reference
        // jQuery Event 처리 코드 내에서
        // this는 이벤트 소스에 대한 문서 객체를 지칭
        alert($(this).text());
    })
}
```

### AJAX

```jsx
function myFunc() {
    $.ajax({
        async: true,
        url: "http://kobis.or.kr/kobisopenapi/webservice/rest/boxoffice/searchDailyBoxOfficeList.json",
        type: "GET",
        timeout: 5000,
        data: {
            key: "da9b8568a54d90fd33fc9a9045f058f9",
            targetDt: "20250218",
            itemPerPage: "10"
        },
        dataType: "json",
        success: function (data) {
            $("h1").text(`1등 영화 제목 : ${data.boxOfficeResult.dailyBoxOfficeList[0].movieNm}`);
            $("input[type=button]").remove();
        },
        error: function (jqXHR, textStatus, errorThrown) {
            console.error(jqXHR.status); // 404
            console.error(textStatus); // "error"
            console.error(errorThrown); // "Not Found"
        }
    })
}
```
