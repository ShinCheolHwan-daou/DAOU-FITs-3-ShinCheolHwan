realloc의 개념에 대해 다시 생각해보고, fopen, fgets, fputs 등을 활용해서 문제를 해결했습니다.

void realloc01() {
    //  1) 요구사항
    //  - 고객은 여러 개의 주식 주문을 할 수 있습니다.
    //  - 각 주문에는 주식 이름, 주문 수량, 주문 가격을 포함합니다.
    //  - 고객은 처음에 주문할 주식의 개수를 입력한 후, 주문을 추가할 때마다 주문 수를 늘릴 수 있습니다.
    //  - realloc() 함수를 사용하여 주문을 추가할 때마다 기존 배열의 크기를 늘려야 합니다.
    //  - 사용자가 입력한 주식 주문 정보를 출력해야 합니다.
    //  - 고객의 주문 내역이 모두 입력된 후, 각 주문에 대해 주식 이름, 수량, 가격을 출력합니다.
    //  2) 처리순서
    //  - 사용자가 주문할 초기 주문 개수를 입력받습니다.
    //  - 주식 주문 내역을 동적 배열로 관리하고, 주문을 추가할 때마다 realloc()을 사용하여 주문수를 늘려야 합니다.
    //  - 각 주문에 대해 주식 이름, 주문 수량, 주문 가격을 입력받고, 주문 정보를 출력합니다.
    //  - 최종적으로 모든 주문 내역을 출력합니다.
    typedef struct {
        char name[300];
        int amount;
        double price;
    } Stock;

    int n;
    printf("초기 주문 개수를 입력하세요");
    scanf("%d", &n);

    Stock *stocks = (Stock *) malloc(n * sizeof(Stock));
    if (stocks == NULL) {
        printf("메모리 할당 실패\n");
        return;
    }

    for (int i = 0; i < n; i++) {
        printf("주식 주문 %d:\n", i + 1);
        printf("주식 이름: ");
        scanf("%s", stocks[i].name);
        printf("주식 수량: ");
        scanf("%d", &stocks[i].amount);
        printf("주식 가격: ");
        scanf("%lf", &stocks[i].price);
    }

    char option;
    int addNum;
    printf("추가로 주문을 입력하시겠습니까?(y / n)");
    getchar();
    scanf("%c", &option);

    if (option == 'y' || option == 'Y') {
        printf("추가할 주문 개수를 입력하세요: ");
        scanf("%d", &addNum);
        Stock *newStocks = (Stock *) realloc(stocks, (n + addNum) * sizeof(Stock));
        if (newStocks == NULL) {
            printf("메모리 할당 실패\n");
            return;
        }
        stocks = newStocks;
        for (int i = n; i < n + addNum; i++) {
            printf("주식 주문 %d:\n", i + 1);
            printf("주식 이름: ");
            scanf("%s", stocks[i].name);
            printf("주식 수량: ");
            scanf("%d", &stocks[i].amount);
            printf("주식 가격: ");
            scanf("%lf", &stocks[i].price);
        }

        n += addNum;
    }

    printf("주식 주문 내역:\n");
    for (int i = 0; i < n; i++) {
        printf("주식 이름: %s, 수량: %d, 가격: %.2lf\n",
               stocks[i].name, stocks[i].amount, stocks[i].price
        );
    }

    free(stocks);
}

void dynamicMemory() {
    // 1. 거래 유형은 현금 거래 (CASH), 카드 거래 (CARD), **가상화폐 거래 (CRYPTO)**로 구분됩니다.
    // 2. 거래 정보를 저장하는 구조체(Transaction)와, 거래 유형에 따라 필요한 데이터를 저장하는 공용체(TransactionData)를 사용합니다.
    // 3. 각 거래의 고유 ID-자동으로 1개씩 증가됨, 거래 유형, 고객 이름을 포함한 거래 정보를 동적 배열로 관리합니다.
    // 4. 거래 유형에 따라 필요한 데이터는 공용체(TransactionData) 내에서 관리됩니다.
    // 현금 거래: 거래 금액(현금)을 저장
    // 카드 거래: 카드 번호, 할부 개월 수를 저장
    // 가상화폐 거래: 지갑 주소, 거래 금액을 저장
    // 5. 거래 내역을 추가, 수정, 삭제하는 기능을 구현합니다.
    // 6. 거래 내역을 출력하는 기능을 구현합니다.
    // 7. 프로그램 종료 시, 동적으로 할당된 메모리를 해제합니다
    typedef enum {
        CASH,
        CARD,
        CRYPTO
    } TransactionType;

    typedef union {
        struct {
            double price;
        } cash;

        struct {
            char cardNumber[50];
            int installment;
            double price;
        } card;

        struct {
            char address[50];
            double price;
        } crypto;
    } TransactionData;

    typedef struct {
        int id;
        char name[300];
        TransactionType type;
        TransactionData data;
    } Transaction;
    int n = 0;
    Transaction *transactions = NULL;

    while (true) {
        printf("1. 거래 추가\n");
        printf("2. 거래 내역 출력\n");
        printf("3. 종료\n");
        int choice;
        printf("선택: ");
        scanf("%d", &choice);

        if (choice == 1) {
            Transaction *newTransactions = (Transaction *) realloc(transactions, (n + 1) * sizeof(Transaction));
            if (newTransactions == NULL) {
                printf("메모리 할당 실패!\n");
                free(transactions);
                return;
            }

            transactions = newTransactions;
            transactions[n].id = n + 1;
            printf("고객 이름: ");
            scanf("%s", transactions[n].name);
            printf("거래 유형 (0: 현금, 1: 카드, 2: 가상화폐):");
            scanf("%d", &transactions[n].type);

            if (transactions[n].type == CASH) {
                printf("금액: ");
                scanf("%lf", &transactions[n].data.cash.price);
            } else if (transactions[n].type == CARD) {
                printf("카드 번호: ");
                scanf("%s", transactions[n].data.card.cardNumber);
                printf("할부 개월 수: ");
                scanf("%d", &transactions[n].data.card.installment);
                printf("카드 거래 금액: ");
                scanf("%lf", &transactions[n].data.card.price);
            } else if (transactions[n].type == CRYPTO) {
                printf("가상화폐 지갑 주소: ");
                scanf("%s", transactions[n].data.crypto.address);
                printf("가상화폐 거래 금액: ");
                scanf("%lf", &transactions[n].data.crypto.price);
            } else {
                printf("잘못된 거래 유형입니다.\n");
            }
            n++;
        } else if (choice == 2) {
            for (int i = 0; i < n; i++) {
                Transaction t = transactions[i];

                if (t.type == CASH) {
                    printf("ID: %d, 고객 이름: %s, 거래 유형: 현금 거래, 금액: %.2lf\n",
                           t.id,
                           t.name,
                           t.data.cash.price
                    );
                } else if (t.type == CARD) {
                    printf("ID: %d, 고객 이름: %s, 거래 유형: 카드 거래, 카드 번호: %s, 할부 개월 수: %d, 금액: %.2lf\n",
                           t.id,
                           t.name,
                           t.data.card.cardNumber,
                           t.data.card.installment,
                           t.data.card.price
                    );
                } else if (t.type == CRYPTO) {
                    printf("ID: %d, 고객 이름: %s, 거래 유형: 가상화폐 거래, 지갑 주소: %s, 금액: %.2lf\n",
                           t.id,
                           t.name,
                           t.data.crypto.address,
                           t.data.crypto.price
                    );
                } else {
                    printf("ID: %d, 고객 이름: %s, 거래 유형: 잘못된 거래 유형입니다.\n",
                           t.id,
                           t.name
                    );
                }
            }
        } else if (choice == 3) {
            printf("종료\n");
            break;
        } else {
            printf("잘못된 입력입니다.\n");
        }
    }

    free(transactions);
}


typedef struct {
    unsigned int is_active: 1; // 활성화 여부 (1비트)
    unsigned int is_vip: 1; // VIP 여부 (1비트)
    unsigned int has_loan: 1; // 대출 여부 (1비트)
    unsigned int over_protection: 1; // 예금 보호 한도 초과 여부 (1비트)
    unsigned int balance_status: 3; // 잔액 상태 (3비트)
} AccountStatusBitField;


void bitField() {
    AccountStatusBitField customer1 = {1, 1, 1, 1, 5};
    printf("계좌 상태:\n");

    };

typedef struct {
    uint8_t flags;
} AccountStatusBitOperation;

void bitOperation() {
    AccountStatusBitOperation customer1;
    customer1.flags = (1 << 0) | (1 << 1) | (1 << 2) | (1 << 3) | (5 << 4);
    // 출력
    printf("계좌 상태:\n");
    printf(" - 활성화: %s\n", (customer1.flags & 0x1) ? "예" : "아니오");
    printf(" - VIP: %s\n", (customer1.flags & 0x2) ? "예" : "아니오");
    printf(" - 대출 여부: %s\n", (customer1.flags & 0x4) ? "있음" : "없음");
    printf(" - 예금 보호 한도 초과: %s\n", (customer1.flags & 0x8)
                                       ? "초과"
                                       : "초과하지 않음");
    const char *balance_labels[] = {
        "0원", "1~10만 원", "10~50만 원", "50~100만 원",
        "100~500만 원", "500~1000만 원", "1000~5000만 원", "5000만 원 이상"
    };
    int balance_status = (customer1.flags & 0x70) >> 4;
    printf(" - 잔액 상태: %s\n", balance_labels[balance_status]);

    // 구조체 크기 확인
    printf("\nAccountStatus 구조체 크기: %llu 바이트\n", sizeof(AccountStatusBitOperation));
}

void optimization() {
    // 1) 요구사항
    // - 사용자 권한을 각각 입력 받아서 권한 4개를 설정하세요.
    // - 권한이 있으면 ‘0’ 없으면 ‘X’로 표시하여 출력하세요.
    // - 각 사용자에게 다음과 같이 4가지 권한이 있다.
    // READ – 읽기권한 1비트
    // WRITE – 쓰기권한 1비트
    // EXECUTE – 실행권한 1비트
    // ADMIN – 관리자권한 1비트
    // 2) 풀이 방식은 비트필드, 비트 연산으로 푸세요
    typedef struct {
        unsigned int read: 1;
        unsigned int write: 1;
        unsigned int execute: 1;
        unsigned int admin: 1;
    } UserPermissionsBitField;
    UserPermissionsBitField permission1;
    int choice;
    printf("사용자 권한을 설정합니다.\n");

    printf("READ 권한을 부여할까요? (1: 예, 0: 아니오)");
    scanf("%d", &choice);
    if (choice != 1 && choice != 0) {
        printf("선택 오류\n");
        return;
    }
    permission1.read = choice;

    printf("WRITE 권한을 부여할까요? (1: 예, 0: 아니오)");
    scanf("%d", &choice);
    if (choice != 1 && choice != 0) {
        printf("선택 오류\n");
        return;
    }
    permission1.write = choice;

    printf("EXECUTE 권한을 부여할까요? (1: 예, 0: 아니오)");
    scanf("%d", &choice);

    if (choice != 1 && choice != 0) {
        printf("선택 오류\n");
        return;
    }
    permission1.execute = choice;

    printf("ADMIN 권한을 부여할까요? (1: 예, 0: 아니오)");
    scanf("%d", &choice);

    if (choice != 1 && choice != 0) {
        printf("선택 오류\n");
        return;
    }
    permission1.admin = choice;

    printf("사용자 권한 상태:\n");
    printf("-READ: %c\n", permission1.read ? 'O' : 'X');
    printf("-WRITE: %c\n", permission1.write ? 'O' : 'X');
    printf("-EXECUTE: %c\n", permission1.execute ? 'O' : 'X');
    printf("-ADMIN: %c\n", permission1.admin ? 'O' : 'X');

    printf("UserPermissions 구조체 크기: %llu 바이트\n", sizeof(UserPermissionsBitField));

    typedef struct {
        uint8_t flags;
    } UserPermissionsBitOperation;

    UserPermissionsBitOperation permission2 = {0};
    printf("READ 권한을 부여할까요? (1: 예, 0: 아니오)");
    scanf("%d", &choice);
    if (choice != 1 && choice != 0) {
        printf("선택 오류\n");
        return;
    }
    permission2.flags |= choice << 3;

    printf("WRITE 권한을 부여할까요? (1: 예, 0: 아니오)");
    scanf("%d", &choice);
    if (choice != 1 && choice != 0) {
        printf("선택 오류\n");
        return;
    }
    permission2.flags |= choice << 2;

    printf("EXECUTE 권한을 부여할까요? (1: 예, 0: 아니오)");
    scanf("%d", &choice);

    if (choice != 1 && choice != 0) {
        printf("선택 오류\n");
        return;
    }
    permission2.flags |= choice << 1;


    printf("ADMIN 권한을 부여할까요? (1: 예, 0: 아니오)");
    scanf("%d", &choice);

    if (choice != 1 && choice != 0) {
        printf("선택 오류\n");
        return;
    }
    permission2.flags |= choice << 0;

    printf("사용자 권한 상태:\n");

    printf("-ADMIN: %c\n", permission2.flags & 0x1 ? 'O' : 'X');

    printf("UserPermissions 구조체 크기: %llu 바이트\n", sizeof(UserPermissionsBitOperation));
}

## 오늘 내가 새롭게 배운 것들
- realloc → 메모리 크기 재할당 (ptr = realloc(ptr, new_size);)
- fopen → 파일 열기 (FILE *fp = fopen("file.txt", "r");)
- fclose → 파일 닫기 (fclose(fp);)
- fgets → 파일/입력에서 한 줄 읽기 (fgets(buffer, size, fp);)
- fputs → 파일/출력에 문자열 쓰기 (fputs("Hello", fp);)
- strstr → 문자열에서 부분 문자열 찾기 (char *pos = strstr(str, "ID:");)
- sscanf → 문자열에서 데이터 추출 (sscanf(str, "%d", &num);)
