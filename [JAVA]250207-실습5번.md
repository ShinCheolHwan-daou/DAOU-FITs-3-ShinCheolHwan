```Java
package javaexam2.company;

public abstract class Employee {
    private String name;
    private int number;
    private String department;
    private int salary;

    // constructor
    public Employee() {

    }

    public Employee(String name, int number, String department, int salary) {
        this.name = name;
        this.number = number;
        this.department = department;
        this.salary = salary;
    }

    // methods
    // getter & setter
    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public int getNumber() {
        return number;
    }

    public void setNumber(int number) {
        this.number = number;
    }

    public String getDepartment() {
        return department;
    }

    public void setDepartment(String department) {
        this.department = department;
    }

    public int getSalary() {
        return salary;
    }

    public void setSalary(int salary) {
        this.salary = salary;
    }

    // others
    public abstract double tax();

    @Override
    public String toString() {
        return String.format("%-10s %-10s %-10d", this.getName(), this.getDepartment(), this.getSalary());
    }
}
```

```Java
package javaexam2.company;

public class Secretary extends Employee implements Bonus {

    public Secretary() {
    }

    public Secretary(String name, int number, String department, int salary) {
        super(name, number, department, salary);
    }

    @Override
    public double tax() {
        return getSalary() * 0.1;
    }

    @Override
    public void incentive(int pay) {
        setSalary(getSalary() + (int) (pay * 0.8));
    }
}
```

```Java
package javaexam2.company;

import java.util.*;

public class Company {
    public static void main(String[] args) {
        Map<Integer, Employee> map = new HashMap<>();
        // 1번의 데이터를 기반으로 객체를 생성 하여 HashMap에 넣는다.
        // HashMap에 넣을 때 키 값은 각 객체의 Number로 한다.
        Employee e1 = new Secretary("Hilery", 1, "secretary", 800);
        Employee e2 = new Sales("Clinten", 2, "sales", 1200);
        map.put(e1.getNumber(), e1);
        map.put(e2.getNumber(), e2);

        // 모든 객체의 기본 정보를 출력 한다(for문을 이용하여 출력 한다.)
        System.out.printf("%-10s %-10s %-10s\n", "name", "department", "salary");
        System.out.println("------------------------------------");
        for (Employee e : map.values()) {
            if (e == null) {
                continue;
            }
            System.out.println(e);
        }

        // 모든 객체에 인센티브 100씩 지급하고 다시 HashMap에 넣는다.
        System.out.println("인센티브 100 지급");
        for (Employee e : map.values()) {
            if (e instanceof Bonus) {
                ((Bonus) e).incentive(100);
            }
        }

        // 모든 객체의 정보와 세금을 출력 한다 (for문을 이용하여 출력 한다.)
        System.out.printf("%-10s %-10s %-10s %-10s\n", "name", "department", "salary", "tax");
        System.out.println("------------------------------------");
        for (Employee e : map.values()) {
            if (e == null) {
                continue;
            }
            System.out.println(e + " " + e.tax());
        }
    }
}
```
