void calloc01() {
    // 동적 할당
    int rows = 3, cols = 4;
    int **ptr = (int **) calloc(rows, sizeof(int *));
    if (ptr == NULL) {
        return;
    }

    for (int i = 0; i < rows; i++) {
        ptr[i] = (int *) calloc(cols, sizeof(int));
        if (ptr[i] == NULL) {
            return;
        }
    }

    // 값 입력
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            ptr[i][j] = i * cols + j;
            printf("%d\t", ptr[i][j]);
        }
        printf("\n");
    }

    // 메모리 해제
    for (int i = 0; i < rows; i++) {
        free(ptr[i]);
    }
    free(ptr);
}

void malloc2() {
    // - 학생 수는 사용자로부터 입력받습니다.
    // - 각 학생은 이름(최대 100자)과 점수(정수)를 가지고 있습니다.
    // - malloc()을 이용하여 학생 수만큼의 메모리를 동적으로 할당합니다.
    // - 학생 정보 입력: 학생의 이름과 점수를 입력받습니다.
    // - 학생 정보 출력: 각 학생의 이름과 점수를 출력합니다.
    // - 메모리 해제: 동적으로 할당된 메모리는 사용 후 free()를 이용해 해제해야 합니다.
    // - 각 학생의 점수는 0~100 사이의 정수여야 하며, 점수 입력 시 유효성 검사를 해야 합니다
    int n;

    typedef struct {
        char name[300];
        int score;
    } Student;

    Student *ptr;
    printf("학생 수를 입력하세요");
    scanf("%d", &n);

    ptr = (Student *) malloc(n * sizeof(Student));

    for (int i = 0; i < n; i++) {
        printf("학생 %d의 이름을 입력해주세요", i + 1);
        scanf("%s", ptr[i].name);

        int score = -1;
        while (true) {
            printf("학생 %d의 점수를 입력하세요 (0~100)", i + 1);
            scanf("%d", &score);

            if (score >= 0 && score <= 100) {
                ptr[i].score = score;
                break;
            }
            printf("잘못된 점수입니다. 다시 입력해주세요.\n");
        }
    }

    printf("입력된 학생의 정보\n");
    for (int i = 0; i < n; i++) {
        printf("학생 %d: %s, 점수: %d\n", i + 1, ptr[i].name, ptr[i].score);
    }

    free(ptr);
}

void malloc4() {
    // - 고객 정보 (이름, 계좌 번호)와 대출 정보 (대출 금액, 대출 이자율, 대출 기간)를 관리합니다.
    // - 고객은 여러 개의 대출을 받을 수 있으며, 각 대출의 정보를 동적으로 할당된 메모리로 관리합니다.
    // - 대출 금액은 양의 정수, 이자율은 0~100 사이의 실수, 대출 기간은 1년 이상이어야 합니다.
    // - 각 고객은 대출을 상환할 수 있습니다. 상환할 때는 상환 금액을 입력받아 대출 잔액을 차감합니다.
    // - 프로그램 종료 전에 각 고객의 대출 잔액을 출력하고, 동적으로 할당된 모든 메모리를 해제해야 합니다.

    // 대출 정보를 저장할 구조체
    typedef struct {
        double loan_amount;
        double interest_rate;
        int loan_period; // 대출 기간 (년)
        double remaining_balance;
    } Loan;

    // 고객 정보를 저장할 구조체
    typedef struct {
        char name[300];
        int account_number;
        int loan_count;
        Loan *loans; // 동적으로 할당된 대출 정보 배열
    } Customer;

    int n;
    printf("고객 수를 입력하세요");
    scanf("%d", &n);

    Customer *customers = (Customer *) malloc(n * sizeof(Customer));
    for (int i = 0; i < n; i++) {
        printf("고객 %d의 이름을 입력하세요", i + 1);
        scanf("%s", customers[i].name);
        printf("고객 %d의 계좌 번호를 입력하세요", i + 1);
        scanf("%d", &customers[i].account_number);
        printf("고객 %d이 받는 대출 수", i + 1);
        scanf("%d", &customers[i].loan_count);

        customers[i].loans = (Loan *) malloc(customers[i].loan_count * sizeof(Loan));
        for (int j = 0; j < customers[i].loan_count; j++) {
            printf("대출 %d\n", j + 1);
            while (true) {
                printf("대출 금액을 입력하세요");
                scanf("%lf", &customers[i].loans[j].loan_amount);

                if (customers[i].loans[j].loan_amount < 0.0) {
                    printf("잘못된 금액입니다. 다시 입력해주세요.\n");
                    continue;
                }
                customers[i].loans[j].remaining_balance = customers[i].loans[j].loan_amount;
                break;
            }
            while (true) {
                printf("대출 이자율을 입력하세요(0~100)");
                scanf("%lf", &customers[i].loans[j].interest_rate);

                if (customers[i].loans[j].interest_rate < 0.0 || customers[i].loans[j].interest_rate > 100) {
                    printf("잘못된 이자율입니다. 다시 입력해주세요.\n");
                    continue;
                }
                break;
            }
            while (true) {
                printf("대출 기간을 입력하세요(1년 이상)");
                scanf("%d", &customers[i].loans[j].loan_period);

                if (customers[i].loans[j].loan_period < 1) {
                    printf("잘못된 기간입니다. 다시 입력해주세요.\n");
                    continue;
                }
                break;
            }
        }

        for (int j = 0; j < customers[i].loan_count; j++) {
            double amount;
            while (true) {
                printf("상환할 금액을 입력하세요 (대출 %d)", j + 1);
                scanf("%lf", &amount);
                if (amount < 0.0) {
                    printf("잘못된 금액입니다. 다시 입력해주세요.\n");
                    continue;
                }
                customers[i].loans[j].remaining_balance -= amount;
                break;
            }
        }
    }


    printf("입력된 고객 및 대출 정보\n");
    for (int i = 0; i < n; i++) {
        printf("고객 %d: %s, 계좌번호: %d\n", i + 1, customers[i].name, customers[i].account_number);
        for (int j = 0; j < customers[i].loan_count; j++) {
            printf("대출 %d: 대출 금액: %.2lf, 이자율: %.2lf%%, 대출 기간 %d년, 대출잔액: %.2lf\n",
                   j + 1,
                   customers[i].loans[j].loan_amount,
                   customers[i].loans[j].interest_rate,
                   customers[i].loans[j].loan_period,
                   customers[i].loans[j].remaining_balance
            );
        }
    }

    // 메모리 초기화
    for (int i = 0; i < n; i++) {
        free(customers[i].loans);
    }
    free(customers);
}

void calloc1() {
    // - 고객 정보 (고객 이름, 계좌 번호, 거래 내역)와 주식 거래 내역 (주식 이름, 거래 수량, 거래 가격)을 동적으로 관리합니다.
    // - 고객은 여러 번 거래를 할 수 있으며, 각 거래 내역은 동적 배열로 관리됩니다.
    // - 프로그램은 사용자가 입력한 고객 수에 맞게 동적으로 메모리 할당하고, 각 고객의 주식 거래 정보를 저장하고 출력합니다.
    // - 주식 거래 내역을 출력할 때, 각 거래는 주식이름, 거래수량, 거래가격을 포함해야 합니다.
    // - 사용자가 종료 시, 동적으로 할당된 메모리는 반드시 free()를 사용하여 해제해야 합니다.
    // - 고객 수는 1명 이상, 100명 이하로 입력됩니다.
    // - 주식 이름은 최대 50자까지 입력됩니다.
    // - 구매 가격, 현재 가격, 보유 수량은 양의 정수로 입력됩니다.
    // - 포트폴리오 내 주식 종목 수는 1개 이상, 50개 이하입니다.
    // - 동적으로 할당된 메모리는 모두 사용 후 반드시 free()를 호출하여 해제해야 합니다.

    typedef struct {
        char name[300];
        int amount;
        double price;
    } Transaction;

    typedef struct {
        char name[300];
        int account;
        int transactionCount;
        Transaction *transactions;
    } Customer;

    int n;
    printf("고객 수를 입력하세요");
    scanf("%d", &n);

    Customer *customers = (Customer *) calloc(n, sizeof(Customer));
    for (int i = 0; i < n; i++) {
        printf("고객 %d의 이름을 입력하세요: ", i + 1);
        scanf("%s", customers[i].name);
        printf("고객 %d의 계좌 번호를 입력하세요: ", i + 1);
        scanf("%d", &customers[i].account);
        printf("고객 %d이 거래 내역 수: ", i + 1);
        scanf("%d", &customers[i].transactionCount);

        customers[i].transactions = (Transaction *) calloc(customers[i].transactionCount, sizeof(Transaction));
        for (int j = 0; j < customers[i].transactionCount; j++) {
            printf("거래 %d\n", j + 1);
            printf("주식 이름을 입력하세요: ");
            scanf("%s", customers[i].transactions[j].name);

            printf("거래 수량을 입력하세요: ");
            scanf("%d", &customers[i].transactions[j].amount);

            printf("거래 가격을 입력하세요: ");
            scanf("%lf", &customers[i].transactions[j].price);
        }
    }

    for (int i = 0; i < n; i++) {
        printf("고객 %d: %s, 계좌번호: %d\n", i + 1, customers[i].name, customers[i].account);
        for (int j = 0; j < customers[i].transactionCount; j++) {
            printf("거래 %d: 주식 이름: %s, 거래 수량: %d, 거래 가격: %.2lf\n",
                   j + 1,
                   customers[i].transactions[j].name,
                   customers[i].transactions[j].amount,
                   customers[i].transactions[j].price
            );
        }
    }

    for (int i = 0; i < n; i++) {
        free(customers[i].transactions);
    }
    free(customers);
}

void calloc2() {
    // - 고객 정보는 이름, 계좌 번호와 포트폴리오를 포함합니다.
    // - 포트폴리오는 여러 주식 종목을 포함하며, 각 주식의 구매 가격, 현재 가격, 보유 수량을 관리합니다.
    // - 각 고객은 여러 종목을 보유할 수 있습니다.
    // - 사용자는 고객을 추가하고, 각 고객의 포트폴리오를 추가 및 수정할 수 있습니다.
    // - 각 고객의 포트폴리오는 동적으로 할당된 배열로 관리되며, 포트폴리오 추가 시 calloc()을 사용하여 초기화됩니다.
    // - 프로그램은 포트폴리오에 있는 각 주식의 총 구매 금액과 현재 가치를 계산하여 출력합니다.
    // - 종료 시에는 동적으로 할당된 메모리를 반드시 해제해야 합니다.
    typedef struct {
        char name[300];
        double buyPrice;
        double curPrice;
        int amount;
    } Stock;

    typedef struct {
        char name[300];
        int account;
        int stockCount;
        Stock *stocks;
    } Customer;

    int n;
    printf("고객 수를 입력하세요");
    scanf("%d", &n);

    Customer *customers = (Customer *) calloc(n, sizeof(Customer));
    for (int i = 0; i < n; i++) {
        printf("고객 %d의 이름을 입력하세요: ", i + 1);
        scanf("%s", customers[i].name);
        printf("고객 %d의 계좌 번호를 입력하세요: ", i + 1);
        scanf("%d", &customers[i].account);
        printf("고객 %d의 포트폴리오에 포함될 종목 수: ", i + 1);
        scanf("%d", &customers[i].stockCount);

        customers[i].stocks = (Stock *) calloc(customers[i].stockCount, sizeof(Stock));
        for (int j = 0; j < customers[i].stockCount; j++) {
            printf("주식 종목 %d\n", j + 1);
            printf("주식 이름을 입력하세요: ");
            scanf("%s", customers[i].stocks[j].name);

            printf("구매 가격을 입력하세요: ");
            scanf("%lf", &customers[i].stocks[j].buyPrice);

            printf("현재 가격을 입력하세요: ");
            scanf("%lf", &customers[i].stocks[j].curPrice);

            printf("보유 주식 수를 입력하세요: ");
            scanf("%d", &customers[i].stocks[j].amount);
        }
    }

    for (int i = 0; i < n; i++) {
        printf("고객 %d: %s, 계좌번호: %d\n", i + 1, customers[i].name, customers[i].account);
        printf("포트폴리오:\n");
        for (int j = 0; j < customers[i].stockCount; j++) {
            printf("주식 종목 %d: %s, 구매 가격: %.2lf, 현재 가격: %.2lf, 보유 수량: %d, 총 구매 금액: %.2lf, 현재 가치: %.2lf\n",
                   j + 1,
                   customers[i].stocks[j].name,
                   customers[i].stocks[j].buyPrice,
                   customers[i].stocks[j].curPrice,
                   customers[i].stocks[j].amount,
                   customers[i].stocks[j].amount * customers[i].stocks[j].buyPrice,
                   customers[i].stocks[j].amount * customers[i].stocks[j].curPrice
            );
        }
    }
    for (int i = 0; i < n; i++) {
        free(customers[i].stocks);
    }
    free(customers);
}

## 오늘 내가 새롭게 배운 것들
- malloc(size) → 지정한 크기만큼 초기화되지 않은 메모리 할당
- calloc(n, size) → n * size 크기의 0으로 초기화된 메모리 할당
- realloc(ptr, size) → 기존 ptr의 메모리를 크기 변경 (주소 변경 가능)
