void addAccount(Customer03 *customer) {
    Account03 *account = &customer->accounts[customer->accountCount];

    printf("추가할 계좌 번호를 입력하세요: \n");
    scanf("%s", account->accountId);
    printf("계좌의 초기 잔액을 입력하세요: \n");
    scanf("%lf", &account->balance);

    customer->accountCount++;
}

void deleteAccount(Customer03 *customer) {
    char accountId[50];
    int deleteIndex = -1;
    printf("삭제할 계좌 번호를 입력하세요: ");
    scanf("%s", accountId);

    for (int i = 0; i < customer->accountCount; i++) {
        if (strcmp(accountId, customer->accounts[i].accountId) == 0) {
            deleteIndex = i;
            break;
        }
    }

    if (deleteIndex == -1) {
        return;
    }

    for (int i = deleteIndex; i < customer->accountCount - 1; i++) {
        customer->accounts[i] = customer->accounts[i + 1];
    }

    customer->accountCount--;
}

void printCustomerInfo(Customer03 *customer) {

}

int ascending(const void *first, const void *second) {
    FixedDeposit *d1 = (FixedDeposit *) first;
    FixedDeposit *d2 = (FixedDeposit *) second;

    struct tm endDate1 = d1->startDate;
    endDate1.tm_mon += d1->depositMonth;
    time_t e1 = mktime(&endDate1);

    struct tm endDate2 = d2->startDate;
    endDate2.tm_mon += d2->depositMonth;
    time_t e2 = mktime(&endDate2);

    if (e1 > e2) {
        return 1;
    } else if (e1 == e2) {
        return 0;
    } else {
        return -1;
    }
}

void sortByMaturityDate(int length, FixedDeposit *fixedDeposits) {
    qsort(fixedDeposits, length, sizeof(FixedDeposit), ascending);
}

bool checkMaturity(FixedDeposit deposit) {
    time_t now;
    localtime(&now); // 현재 시간 정보를 tm 구조체로 변환

    struct tm finalTime = deposit.startDate;
    finalTime.tm_mon += deposit.depositMonth;
    time_t final = mktime(&finalTime);

    if (now >= final) {
        return true;
    } else {
        return false;
    }
}

void displayDeposits(int length, FixedDeposit fixedDeposits[MAX_DEPOSITS]) {
    printf("[정기 예금 목록 (만기일 빠른 순)]\n");
    for (int i = 0; i < length; i++) {
        FixedDeposit deposit = fixedDeposits[i];
        struct tm endDate = deposit.startDate;
        endDate.tm_mon += deposit.depositMonth;
        mktime(&endDate);
        printf("%d %10s | 계좌: %10s | 예금 시작: %d-%02d-%02d | 만기 : %d-%02d-%02d | 금액: %10.2lf | %10s\n",
               i + 1,
               deposit.name,
               deposit.accountNumber,
               deposit.startDate.tm_year + 1900,
               deposit.startDate.tm_mon + 1,
               deposit.startDate.tm_mday,
               endDate.tm_year + 1900,
               endDate.tm_mon + 1,
               endDate.tm_mday,
               deposit.amount,
               checkMaturity(deposit) ? "만기 도래" : "만기 대기"
        );
    }
}

void calculateAverage(Stock *stock) {
    double avg = 0.0;

    for (int i = 0; i < sizeof(stock->endPrice) / sizeof(double); i++) {
        avg += stock->endPrice[i];
    }

    avg = avg / (sizeof(stock->endPrice) / sizeof(double));

    stock->avgPrice = avg;
}

void calculateVolatility(Stock *stock) {
    double vol = 0.0;
    for (int i = 0; i < sizeof(stock->endPrice) / sizeof(double); i++) {
        vol += pow(stock->endPrice[i] - stock->avgPrice, 2);
    }

    vol = vol / (sizeof(stock->endPrice) / sizeof(double));
    vol = sqrt(vol);
    stock->volatility = vol;
}

char *categorizeRisk(Stock *stock) {
    if (stock->volatility < 5) {
        return "낮음";
    } else if (stock->volatility < 10) {
        return "보통";
    } else {
        return "높음";
    }
}

void displayPortfolio(Stock stocks[]) {
    for (int i = 0; i < STOCKS; i++) {
        printf("종목: %s (%s) | 보유 주식: %d | 변동성: %.2lf | 위험 수준: %s\n",
               stocks[i].name,
               stocks[i].ticker,
               stocks[i].amount,
               stocks[i].volatility,
               categorizeRisk(&stocks[i])
        );
    }

## 오늘 내가 새롭게 배운 것들
구조체(struct): 여러 개의 다른 타입의 변수를 하나의 단위로 묶는 사용자 정의 데이터 타입.<br>
열거형(enum): 관련된 상수들의 집합을 정의하는 데이터 타입.<br>
공용체(union): 구조체와 유사하지만, 모든 멤버가 동일한 메모리 공간을 공유하는 데이터 타입.<br>
