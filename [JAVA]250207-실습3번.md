package javaexam2;

import java.util.*;

public class Test05 {
    public static void main(String[] args) {
        Map<Integer, Integer> map = new HashMap<>();
        double sum = 0.0;
        for (int i = 0; i < 10; i++) {
            // HashMap에 1~100까지의 정수를 랜덤하게 10개 셋팅
            map.put(i, (int) (Math.random() * 100 + 1));
        }
        for (int i = 0; i < 10; i++) {
            System.out.print(map.get(i) + " ");
            // HashMap 데이터를 화면에 출력
        }
        System.out.print("\n");

        for (int i = 0; i < 10; i++) {
            // HashMap 데이터의 합계를 계산
            sum += map.get(i);
        }
        // 합계와 평균을 화면에 출력
        System.out.printf("합계: %.2f\n", sum);
        System.out.printf("평균: %.2f\n", sum / map.size());
    }
}
