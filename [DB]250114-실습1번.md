1. 테이블 생성한 "명령어 스크립트 파일" 제출
```SQL
CREATE TABLE "CUSTOMER_INFO" (
    "CUSTOMER_ID" NUMBER PRIMARY KEY,
    "NAME" VARCHAR2(50),
    "SURNAME" VARCHAR2(50),
    "REGISTRATION_DATE" DATE,
    "CUSTOMER_GRADE_ID" NUMBER,
    "MARKETING_INFORMATION" NUMBER(1),
    "PERSONAL_INFORMATION" NUMBER(1)
);

CREATE TABLE "CUSTOMER_GRADES" (
    "CUSTOMER_GRADE_ID" NUMBER PRIMARY KEY,
    "CUSTOMER_GRADE_NAME" VARCHAR2(50)
);

CREATE TABLE "INVESTMENT_PORTFOLIO" (
    "INVESTMENT_PORTFOLIO_ID" NUMBER PRIMARY KEY,
    "CUSTOMER_ID" NUMBER,
    "STOCK_ID" NUMBER,
    "AMOUNT" NUMBER,
    "AVERAGE_PRICE" NUMBER
);

CREATE TABLE "PROFIT_REPORTS" (
    "PROFIT_REPORT_ID" NUMBER PRIMARY KEY,
    "CUSTOMER_ID" NUMBER,
    "START_DATE" DATE,
    "END_DATE" DATE,
    "START_AMOUNT" NUMBER,
    "END_AMOUNT" NUMBER
);

CREATE TABLE "ACCOUNT_INFO" (
    "ACCOUNT_ID" NUMBER PRIMARY KEY,
    "CUSTOMER_ID" NUMBER,
    "ACCOUNT_NUMBER" VARCHAR2(50),
    "BALANCE" NUMBER,
    "ACCOUNT_STATUS" NUMBER,
    "OPENING_DATE" TIMESTAMP,
    "CLOSING_DATE" TIMESTAMP
);

CREATE TABLE "STOCK_INFO" (
    "STOCK_ID" NUMBER PRIMARY KEY,
    "STOCK_NAME" VARCHAR2(100),
    "LISTING_DATE" DATE,
    "EPS" NUMBER,
    "BPS" NUMBER,
    "PER" NUMBER,
    "PBR" NUMBER,
    "ROE" NUMBER,
    "DIVIEND_YIELD" NUMBER,
    "CEO" VARCHAR2(100),
    "CREDIT_RATING" NUMBER 
);

CREATE TABLE "STOCK_DIVIENDS" (
    "STOCK_DIVIEND_ID" NUMBER PRIMARY KEY,
    "DIVIEND_DATE" DATE,
    "DIVIEND_AMOUNT" NUMBER,
    "STOCK_ID" NUMBER,
    "ACCOUNT_ID" NUMBER
);

CREATE TABLE "STOCK_PRICE_INFO" (
    "STOCK_ID" NUMBER PRIMARY KEY,
    "CURRENT_PRICE" NUMBER,
    "PREVIOUS_CLOSING_PRICE" NUMBER,
    "OPENING_PRICE" NUMBER,
    "HIGH_PRICE" NUMBER,
    "LOW_PRICE" NUMBER,
    "TRADING_VOLUME" NUMBER,
    "TRADING_VALUE" NUMBER
);

CREATE TABLE "SECURITIES_ORDERS" (
    "ORDER_ID" NUMBER PRIMARY KEY,
    "ACCOUNT_ID" NUMBER,
    "STOCK_ID" NUMBER,
    "ORDER_AMOUNT" NUMBER,
    "COMPLETE_AMOUNT" NUMBER,
    "ORDER_DATE" TIMESTAMP,
    "COMPLETED_DATE" TIMESTAMP,
    "ORDER_TYPE" NUMBER,
    "PRICE" NUMBER
);

CREATE TABLE "SECURITIES_TRADES" (
    "TRADE_ID" NUMBER PRIMARY KEY,
    "BUY_ORDER_ID" NUMBER,
    "SELL_ORDER_ID" NUMBER,
    "TRADE_AMOUNT" NUMBER,
    "TRADE_DATE" TIMESTAMP,
    "TRADE_PRICE" NUMBER
);

CREATE TABLE "TRANSACTION_FEES" (
    "TRANSACTION_FEE_ID" NUMBER PRIMARY KEY,
    "ORDER_ID" NUMBER,
    "FEE_AMOUNT" NUMBER
);
```
