# JDBC 연습
```Java
package first;

// 책 제목에 대한 키워드를 입력받아서
// 책 제목에 해당 키워드를 포함하고 있는 책을 검색한 다음
// 해당 책에 대한 책 제목, 저자, 가격 정보를 출력하세요!

import java.util.*;
import java.sql.*;

public class SecondConnect {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("검색어를 입력해주세요: ");
        String keyword = sc.nextLine();

        try {
            // 1. JDBC Driver Loading
            Class.forName("oracle.jdbc.driver.OracleDriver");

            // 2. Database 연결
            String JDBC_URL = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##JAVA_DEV";
            String pw = "1234";
            Connection con = DriverManager.getConnection(JDBC_URL, id, pw);

            // 3. SQL 작성
            String sql = "SELECT BTITLE, BAUTHOR, BPRICE FROM BOOK WHERE BTITLE LIKE ?";

            // 4. 쿼리 실행
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "%" + keyword + "%");
            ResultSet rs = pstmt.executeQuery();

            // 5. 결과 처리
            while (rs.next()) {
                System.out.printf("%s %s %s\n",
                        rs.getString("BTITLE"),
                        rs.getString("BAUTHOR"),
                        rs.getString("BPRICE")
                );
            }

            // 6. 리소스 해제
            rs.close();
            pstmt.close();
            con.close();

        } catch (Exception e) {
            System.out.println(e.getMessage());
        }
    }
}
```
