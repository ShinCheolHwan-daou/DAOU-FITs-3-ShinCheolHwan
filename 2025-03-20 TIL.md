void transfer() {
    FILE *file = fopen(FILE_NAME01, "r");
    if (file == NULL) {
        printf("File could not be opened\n");
        return;
    }

    int fromId, toId;
    double amount;
    printf("송금할 계좌 ID: ");
    scanf("%d", &fromId);
    printf("출금할 계좌 ID: ");
    scanf("%d", &toId);
    printf("이체 금액: ");
    scanf("%lf", &amount);


    if (fromId == toId) {
        printf("송금 / 수취 ID가 같습니다.\n");
        fclose(file);
        return;
    }


    char line[1024];
    Account01 account = {-1, "", -1};
    Account01 fromAccount = {-1, "", -1};
    Account01 toAccount = {-1, "", -1};

    while (fgets(line, sizeof(line), file)) {
        sscanf(line, "%d %s %lf", &account.id, account.name, &account.balance);
        if (account.id == toId) {
            memcpy(&toAccount, &account, sizeof(Account01));
        }

        if (account.id == fromId) {
            memcpy(&fromAccount, &account, sizeof(Account01));
        }
    }

    if (fromAccount.id == -1) {
        printf("송금 ID가 존재하지 않는 ID입니다.\n");
        fclose(file);
        return;
    }

    if (toAccount.id == -1) {
        printf("수취 ID가 존재하지 않는 ID입니다.\n");
        fclose(file);
        return;
    }

    if (fromAccount.balance < amount) {
        printf("송금 계좌에 잔액이 부족합니다.\n");
        fclose(file);
        return;
    }
    fseek(file, 0, SEEK_SET);
    FILE *temp = fopen(TEMP_FILE01, "w");
    if (temp == NULL) {
        printf("File could not be opened\n");
        return;
    }
    while (fgets(line, sizeof(line), file)) {
        sscanf(line, "%d %s %lf", &account.id, account.name, &account.balance);
        if (account.id == fromId) {
            fprintf(temp, "%d %s %lf\n", account.id, account.name, account.balance - amount);
        } else if (account.id == toId) {
            fprintf(temp, "%d %s %lf\n", account.id, account.name, account.balance + amount);
        } else {
            fprintf(temp, "%d %s %lf\n", account.id, account.name, account.balance);
        }
    }
    fclose(temp);

    fclose(file);
    if (remove(FILE_NAME01) != 0) {
        printf("File could not be removed\n");
        return;
    }
    rename(TEMP_FILE01, FILE_NAME01);
    printf(">> %d번 계좌에서 %d번 계좌로 %.2lf원이 송금되었습니다.\n", fromAccount.id, toAccount.id, amount);
}

typedef struct person {
    char name[7];

void displayUsers(FILE *fp, int size) {
    PERSON user;
    puts("---------------------------------");
    puts("번호 이름 성별 나이");
    puts("---------------------------------");
    rewind(fp); // 파일의 시작위치로 이동

    }
    puts("---------------------------------");
}

void searchUser(FILE *fp, int size) {
    char searchName[7];
    PERSON user;
    int found = 0;

    scanf("%s", searchName);
    puts("---------------------------------");
    puts("번호 이름 성별 나이");
    puts("---------------------------------");
    rewind(fp); // 파일의 시작위치로 이동
    for (int i = 0; i < size; i++) {
        fread(&user, sizeof(PERSON), 1, fp);
        if (strcmp(user.name, searchName) == 0) {
            printf("%3d %-8s %-3s %4d\n", i + 1, user.name, user.gender, user.age);
            found = 1;
        }
    }

    puts("---------------------------------");
}

    PERSON user;

    scanf("%s", user.gender);
    printf("나이 : ");
    scanf("%d", &user.age);
    fseek(fp, 0, SEEK_END); // 파일 끝으로 이동
    fwrite(&user, sizeof(PERSON), 1, fp);
    printf("새 사용자가 삽입되었습니다.\n");
}

void deleteUser(FILE *fp, int *size) {
    int no;
    PERSON *users = (PERSON *) malloc(*size * sizeof(PERSON));
    rewind(fp);
    fread(users, sizeof(PERSON), *size, fp);
    printf("삭제할 사용자의 번호는? ");
    scanf("%d", &no);
    if (no < 1 || no > *size) {
        printf("잘못된 번호입니다.\n");
    } else {
        // 삭제 후 사용자 수 줄이기
        for (int i = no - 1; i < *size - 1; i++) {
            users[i] = users[i + 1];
        }
        (*size)--; // 사용자 수 감소
        // 파일을 새로 쓰기
        freopen("user.bin", "wb", fp);
        fwrite(users, sizeof(PERSON), *size, fp);
        printf("사용자가 삭제되었습니다.\n");
    }
    free(users);
}
