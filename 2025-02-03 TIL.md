## 어떤 문제가 있었는지
프로젝트로 애널리스트 분석 자료와 주식에 대해 토론할 수 있는 DB 프로젝트에서 구현 쿼리를 작성했습니다.
```mysql
-- 1. 게시글 작성
INSERT INTO BOARDS(BOARD_ID, TITLE, CONTENT, CREATE_AT, USER_ID, STOCK_ID)
VALUES(BOARDS_SEQUENCE.NEXTVAL, '이 주식 좋아요~', '무조건 오름', SYSDATE, '5', 1);

-- 2. 게시글 조회
-- 기본
SELECT *
FROM (
SELECT 
        ROW_NUMBER() OVER (ORDER BY B.CREATE_AT DESC) AS IDX,
        B.BOARD_ID,
        CASE
            WHEN B.DELETE_AT IS NULL THEN B.TITLE
            ELSE '삭제한 게시글'
        END AS TITLE,
        CASE
            WHEN B.DELETE_AT IS NULL THEN B.CONTENT
            ELSE '-'
        END AS CONTENT,
        B.CREATE_AT,
        B.USER_ID,
        CASE
            WHEN U.DELETE_AT IS NULL THEN U.NICKNAME 
            ELSE '탈퇴한 회원'
        END AS NICKNAME,
        B.STOCK_ID,
        S.NAME AS STOCK_NAME,
        (SELECT COUNT(*) FROM REPLIES R WHERE R.BOARD_ID = B.BOARD_ID) AS REPLY_COUNT
    FROM BOARDS B, USERS U, STOCKS S
    WHERE B.USER_ID = U.USER_ID 
        AND B.STOCK_ID = S.STOCK_ID
)
WHERE IDX BETWEEN 11 AND 20;
CREATE INDEX IDX_BOARDS_STOCK_ID ON BOARDS(STOCK_ID);
CREATE INDEX IDX_BOARDS_USER_ID ON BOARDS(USER_ID);
CREATE INDEX IDX_REPLIES_BOARD_ID ON REPLIES(BOARD_ID);
DROP INDEX IDX_BOARDS_STOCK_ID;
DROP INDEX IDX_BOARDS_USER_ID;
DROP INDEX IDX_REPLIES_BOARD_ID;
SELECT * FROM TABLE(DBMS_XPLAN.DISPLAY_CURSOR(NULL, NULL, 'ALLSTATS LAST'));

-- 게시글 종목별 조회
SELECT *
FROM (
    SELECT 
        ROW_NUMBER() OVER (ORDER BY B.CREATE_AT DESC) AS IDX,
        B.BOARD_ID,
        CASE
            WHEN B.DELETE_AT IS NULL THEN B.TITLE
            ELSE '삭제한 게시글'
        END AS TITLE,
        CASE
            WHEN B.DELETE_AT IS NULL THEN B.CONTENT
            ELSE '-'
        END AS CONTENT,
        B.CREATE_AT,
        B.USER_ID,
        CASE
            WHEN U.DELETE_AT IS NULL THEN U.NICKNAME 
            ELSE '탈퇴한 회원'
        END AS NICKNAME,
        B.STOCK_ID,
        S.NAME AS STOCK_NAME,
        (SELECT COUNT(*) FROM REPLIES R WHERE R.BOARD_ID = B.BOARD_ID) AS REPLY_COUNT
    FROM BOARDS B, USERS U, STOCKS S
    WHERE B.USER_ID = U.USER_ID 
        AND B.STOCK_ID = S.STOCK_ID
        AND B.STOCK_ID = 4
)
WHERE IDX BETWEEN 1 AND 10;

-- 게시글 유저별 조회
SELECT *
FROM (
SELECT 
        ROW_NUMBER() OVER (ORDER BY B.CREATE_AT DESC) AS IDX,
        B.BOARD_ID,
        CASE
            WHEN B.DELETE_AT IS NULL THEN B.TITLE
            ELSE '삭제한 게시글'
        END AS TITLE,
        CASE
            WHEN B.DELETE_AT IS NULL THEN B.CONTENT
            ELSE '-'
        END AS CONTENT,
        B.CREATE_AT,
        B.USER_ID,
        CASE
            WHEN U.DELETE_AT IS NULL THEN U.NICKNAME 
            ELSE '탈퇴한 회원'
        END AS NICKNAME,
        B.STOCK_ID,
        S.NAME AS STOCK_NAME,
        (SELECT COUNT(*) FROM REPLIES R WHERE R.BOARD_ID = B.BOARD_ID) AS REPLY_COUNT
    FROM BOARDS B, USERS U, STOCKS S
    WHERE B.USER_ID = U.USER_ID 
        AND B.STOCK_ID = S.STOCK_ID
        AND U.USER_ID = 2
)
WHERE IDX BETWEEN 1 AND 10;


-- 3. 게시글 하위 댓글 조회
SELECT *
FROM (
    SELECT 
        ROW_NUMBER() OVER (ORDER BY R.CREATE_AT) AS IDX,
        R.REPLY_ID,
        CASE
            WHEN R.DELETE_AT IS NULL THEN R.CONTENT 
            ELSE '삭제한 댓글'
        END AS CONTENT,
        R.CREATE_AT,
        R.BOARD_ID,
        R.USER_ID,
        CASE
            WHEN U.DELETE_AT IS NULL THEN U.NICKNAME 
            ELSE '탈퇴한 회원'
        END AS NICKNAME
    FROM REPLIES R, USERS U
    WHERE R.USER_ID = U.USER_ID
        AND R.BOARD_ID = 5
)
WHERE IDX BETWEEN 1 AND 10;


-- 고객 관심사 파악
-- 1. 한달 동안 가장 많은 게시글이 만들어진 종목들

SELECT *
FROM (
    SELECT 
        ROW_NUMBER() OVER(ORDER BY COUNT(*) DESC) AS IDX,
        S.STOCK_ID,
        S.NAME,
        COUNT(*) AS BOARD_COUNT
    FROM BOARDS B, STOCKS S
    WHERE B.STOCK_ID = S.STOCK_ID
        AND B.CREATE_AT BETWEEN SYSDATE - INTERVAL '1' MONTH AND SYSDATE
    GROUP BY S.STOCK_ID, S.NAME
)
WHERE IDX <= 5;


-- 2. 한달 동안 가장 많은 댓글이 달린 게시글들
SELECT * 
FROM (
    SELECT 
        ROW_NUMBER() OVER (ORDER BY COUNT(*) DESC) AS IDX,
        R.BOARD_ID,
        COUNT(*) AS REPLY_COUNT
    FROM REPLIES R
    WHERE R.CREATE_AT BETWEEN SYSDATE - INTERVAL '1' MONTH AND SYSDATE
    GROUP BY R.BOARD_ID
) T, BOARDS B
WHERE T.BOARD_ID = B.BOARD_ID
    AND T.IDX <= 5;


-- 3. 한달 동안 가장 많은 댓글이 달린 유저들
SELECT *
FROM (
    SELECT 
        ROW_NUMBER() OVER (ORDER BY COUNT(*) DESC) AS IDX,
        U.USER_ID,
        U.NICKNAME,
        COUNT(*) AS REPLY_COUNT
    FROM USERS U, BOARDS B, REPLIES R
    WHERE U.USER_ID = B.USER_ID
        AND B.BOARD_ID = R.BOARD_ID
        AND R.CREATE_AT BETWEEN SYSDATE - INTERVAL '1' MONTH AND SYSDATE
    GROUP BY U.USER_ID, U.NICKNAME    
)
WHERE IDX <= 5;
```
