package assignment;

import java.util.*;

public class Test05 {
    public static void main(String[] args) {
        int[] nums = new int[3];
        for (int i = 0; i < 3; i++) {
            nums[i] = Integer.parseInt(args[i]);
        }
        System.out.println("최대값: " + Arrays.stream(nums).max().getAsInt());
        System.out.println("최소값: " + Arrays.stream(nums).min().getAsInt());
    }
}
