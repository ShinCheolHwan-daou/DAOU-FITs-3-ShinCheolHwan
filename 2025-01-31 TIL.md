프로젝트로 애널리스트 분석 자료와 주식에 대해 토론할 수 있는 ERD를 제작하고 TABLE을 만드는 SQL과 DUMMY DATA를 만드는 SQL을 구현했습니다. 아래는 해당 TABLE을 만드는 SQL문입니다.
```SQL
CREATE TABLE "REPLIES" (
	"REPLY_ID" NUMBER PRIMARY KEY,
	"CONTENT" VARCHAR2(3000),
	"CREATE_AT" DATE,
	"DELETE_AT" DATE,
	"BOARD_ID" NUMBER,
	"USER_ID" VARCHAR2(100)
);

CREATE TABLE "ANALYSTS" (
	"ANALYST_ID" NUMBER PRIMARY KEY,
	"NAME" VARCHAR2(50),
	"EDUCATION" VARCHAR2(100),
	"EXPERIENCE" VARCHAR2(100),
	"POSITION" VARCHAR2(50),
	"PHONE_NUMBER" VARCHAR2(50),
	"EMAIL" VARCHAR2(100),
	"IMAGE_PATH" VARCHAR2(200),
	"CREATE_AT" DATE,
	"DELETE_AT" DATE,
	"DEPARTMENT_ID" NUMBER,
	"IS_MANAGER" NUMBER
);

CREATE TABLE "ANALYST_SPECIALTIES" (
	"ANALYST_ID" NUMBER,
	"SPECIALTY_ID" NUMBER
);

CREATE TABLE "USERS" (
	"USER_ID" VARCHAR2(100) PRIMARY KEY,
	"PASSWORD" VARCHAR2(100),
	"NAME" VARCHAR2(50),
	"NICKNAME" VARCHAR2(20),
	"PHONE_NUMBER" VARCHAR2(50),
	"CREATE_AT" DATE,
	"DELETE_AT" DATE
);

CREATE TABLE "BOARDS" (
	"BOARD_ID" NUMBER PRIMARY KEY,
	"TITLE" VARCHAR2(150),
	"CONTENT" VARCHAR2(3000),
	"CREATE_AT" DATE,
	"DELETE_AT" DATE,
	"USER_ID" VARCHAR2(100),
	"STOCK_ID" NUMBER
);

CREATE TABLE "REPORTS" (
	"REPORT_ID" NUMBER PRIMARY KEY,
	"TITLE" VARCHAR2(300),
	"FILE_PATH" VARCHAR2(200),
	"VIEW_COUNT" NUMBER,
	"RATING" NUMBER,
	"CREATE_AT" DATE,
	"DELETE_AT" DATE,
	"ANALYST_ID" NUMBER,
	"REPORT_CATEGORY_ID" NUMBER
);

CREATE TABLE "REPORT_CATEGORIES" (
	"REPORT_CATEGORY_ID" NUMBER PRIMARY KEY,
	"NAME" VARCHAR2(50)
);

CREATE TABLE "CLIPPINGS" (
	"REPORT_ID" NUMBER,
	"USER_ID" VARCHAR2(100)
);

CREATE TABLE "DEPARTMENTS" (
	"DEPARTMENT_ID" NUMBER PRIMARY KEY,
	"NAME" VARCHAR2(50),
	"INTRODUCTION" LONG
);

CREATE TABLE "SPECIALTIES" (
	"SPECIALTY_ID" NUMBER PRIMARY KEY,
	"NAME" VARCHAR2(15)
);

CREATE TABLE "STOCKS" (
	"STOCK_ID" NUMBER PRIMARY KEY,
	"NAME" VARCHAR2(50)
);

ALTER TABLE "ANALYST_SPECIALTIES" ADD CONSTRAINT "PK_ANALYST_SPECIALTIES" PRIMARY KEY (
	"ANALYST_ID",
	"SPECIALTY_ID"
);

ALTER TABLE "CLIPPINGS" ADD CONSTRAINT "PK_CLIPPINGS" PRIMARY KEY (
	"REPORT_ID",
	"USER_ID"
);
```
