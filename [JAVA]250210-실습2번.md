# JDBC Transaction 연습
```Java
package first;

import java.sql.*;

public class TransactionTest {
    public static void main(String[] args) {
        try {
            // 1. JDBC Driver loading
            Class.forName("oracle.jdbc.driver.OracleDriver");

            // 2. Database connect
            String JDBC_URL = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##JAVA_DEV";
            String pw = "1234";
            Connection con = DriverManager.getConnection(JDBC_URL, id, pw);
            // 기본적으로 autoCommit mode로 Connection 생성되지만
            // transaction 처리를 위해 con.setAutoCommit(false)로 설정
            con.setAutoCommit(false);

            // 3. sql 작성
            String sql = "INSERT INTO MEMBERTBL(MEMBER_ID, MEMBER_NAME) VALUES(?,?)";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "2");
            pstmt.setString(2, "신사임당");

            // 4. SQL 실행
            int result = pstmt.executeUpdate();

            // 5. SQL 결과 처리
            if (result == 1) {
                System.out.println("잘 처리되었습니다.");
            }
            // 만약 connection이 정상종료되면 con.commit()으로 간주되지만
            // 명시적으로 con.commit() or con.rollback() 권장
            // con.commit(); (2, 신사임당) 데이터 저장
            // con.rollback(); (2, 신사임당) 데이터 롤백
            con.commit();

            // 6. 리소스 해제
            pstmt.close();
            con.close();

        } catch (Exception e) {
            System.out.println(e.getMessage());
        }
    }
}

```
